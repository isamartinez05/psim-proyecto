% Sección 1
\section{Problema}
\begin{frame}{Problema}
    \begin{itemize}
        \item La delimitación del campo pulmonar precede a toda medición cuantitativa.
        \item Su estabilidad depende del detector, la exposición y los hallazgos del parénquima.
        \item Si la máscara se desplaza, los descriptores morfológicos heredan esa variabilidad.
    \end{itemize}
\end{frame}

% Sección 2
\section{Pregunta y objetivos}
\begin{frame}{Pregunta de investigación}
    \begin{block}{}
        ¿En qué medida las estrategias de realce, umbralización y morfología
        modifican la estabilidad de los descriptores morfológicos del campo
        pulmonar respecto a la máscara de referencia?
    \end{block}
\end{frame}

\begin{frame}{Objetivos específicos}
    \begin{enumerate}
        \item Caracterizar digitalización y adquisición de ambos conjuntos.
        \item Comparar estrategias de normalización y reducción de ruido.
        \item Comparar estrategias de segmentación contra las máscaras de referencia.
        \item Cuantificar el desplazamiento de cada descriptor.
        \item Analizar diferencias entre grupos y documentar limitaciones.
    \end{enumerate}
\end{frame}

% Sección 3
\section{Datos y métodos}
